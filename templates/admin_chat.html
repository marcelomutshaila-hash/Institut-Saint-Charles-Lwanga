import sqlite3 from datetime import datetime from flask import Flask, render_template, request, jsonify app = Flask(__name__) # Initialisation de la base de données def init_db(): conn = sqlite3.connect('database.db') cursor = conn.cursor() cursor.execute(''' CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, sender TEXT, message TEXT, datetime_str TEXT, likes INTEGER DEFAULT 0, dislikes INTEGER DEFAULT 0 ) ''') conn.commit() conn.close() init_db() # Route pour récupérer les messages @app.route('/api/get_messages') def get_messages(): conn = sqlite3.connect('database.db') conn.row_factory = sqlite3.Row cursor = conn.cursor() cursor.execute('SELECT * FROM messages ORDER BY id DESC') rows = cursor.fetchall() conn.close() messages = [dict(row) for row in rows] return jsonify(messages) # Route pour envoyer un message @app.route('/api/send_message', methods=['POST']) def send_message(): data = request.json sender = data.get('sender', 'Visiteur') message = data.get('message', '') if not message.strip(): return jsonify({'error': 'Message vide'}), 400 datetime_str = datetime.now().strftime('%d/%m/%Y à %H:%M') conn = sqlite3.connect('database.db') cursor = conn.cursor() cursor.execute(''' INSERT INTO messages (sender, message, datetime_str, likes, dislikes) VALUES (?, ?, ?, 0, 0) ''', (sender, message, datetime_str)) conn.commit() conn.close() return jsonify({'success': True}) # Route pour supprimer un message (Réservé Admin) @app.route('/api/delete_message/', methods=['DELETE']) def delete_message(msg_id): conn = sqlite3.connect('database.db') cursor = conn.cursor() cursor.execute('DELETE FROM messages WHERE id = ?', (msg_id,)) conn.commit() conn.close() return jsonify({'success': True}) # Route pour les Likes @app.route('/api/like_message/', methods=['POST']) def like_message(msg_id): conn = sqlite3.connect('database.db') cursor = conn.cursor() cursor.execute('UPDATE messages SET likes = likes + 1 WHERE id = ?', (msg_id,)) conn.commit() conn.close() return jsonify({'success': True}) # Route pour les Dislikes @app.route('/api/dislike_message/', methods=['POST']) def dislike_message(msg_id): conn = sqlite3.connect('database.db') cursor = conn.cursor() cursor.execute('UPDATE messages SET dislikes = dislikes + 1 WHERE id = ?', (msg_id,)) conn.commit() conn.close() return jsonify({'success': True})
